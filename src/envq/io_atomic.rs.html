1//! Raw byte file I/O and atomic replacement helpers.
2
3use std::fs;
4#[cfg(unix)]
5use std::fs::File;
6use std::io::{self, Write};
7use std::path::{Path, PathBuf};
8
9use tempfile::Builder;
10
11/// Reads a file as raw bytes.
12pub fn read_bytes(path: &Path) -> io::Result<Vec<u8>> {
13    fs::read(path)
14}
15
16/// Writes bytes through a same-directory temporary file and atomically replaces `path`.
17pub fn write_bytes_atomic(path: &Path, bytes: &[u8]) -> io::Result<()> {
18    let directory = containing_directory(path);
19    let existing_permissions = existing_permissions(path)?;
20    let file_name = path
21        .file_name()
22        .and_then(|name| name.to_str())
23        .unwrap_or("envq");
24    let prefix = format!(".{file_name}.");
25    let mut temporary = Builder::new()
26        .prefix(&prefix)
27        .suffix(".tmp")
28        .tempfile_in(&directory)?;
29
30    if let Some(permissions) = existing_permissions {
31        temporary.as_file().set_permissions(permissions)?;
32    }
33
34    temporary.write_all(bytes)?;
35    temporary.as_file_mut().flush()?;
36    temporary.as_file().sync_all()?;
37
38    // `persist` renames over the path itself. For symlinks, this replaces the
39    // symlink with a regular file instead of following it.
40    temporary
41        .persist(path)
42        .map_err(|error| io::Error::new(error.error.kind(), error.error))?;
43    fsync_directory(&directory);
44    Ok(())
45}
46
47/// Returns the directory used for same-directory temporary files.
48#[must_use]
49pub fn containing_directory(path: &Path) -> PathBuf {
50    match path.parent() {
51        Some(parent) if !parent.as_os_str().is_empty() => parent.to_path_buf(),
52        _ => PathBuf::from("."),
53    }
54}
55
56fn existing_permissions(path: &Path) -> io::Result<Option<fs::Permissions>> {
57    // `metadata` follows symlinks, matching the reference behavior for mode
58    // preservation before the symlink path itself is replaced.
59    match fs::metadata(path) {
60        Ok(metadata) => Ok(Some(metadata.permissions())),
61        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(None),
62        Err(error) => Err(error),
63    }
64}
65
66#[cfg(unix)]
67fn fsync_directory(directory: &Path) {
68    let Ok(file) = File::open(directory) else {
69        return;
70    };
71    // Directory fsync is best effort; replacement success is already reported.
72    let _ = file.sync_all();
73}
74
75#[cfg(not(unix))]
76fn fsync_directory(_directory: &Path) {}
