1//! Clap command metadata used for validation and drift guards.
2//!
3//! User-visible help and diagnostics are still emitted by compatibility code
4//! instead of raw clap output.

6use clap::builder::PossibleValuesParser;

11pub(super) const COMPLETION_SHELLS: &[&str] = &["bash", "zsh", "fish", "powershell", "pwsh"];
12#[cfg(test)]
13pub(super) const MUTATION_OUTPUT_OPTIONS: &[&str] = &["stdout", "diff"];

17/// Full command surface used by tests to detect drift from documented commands.
18#[allow(dead_code)]
19pub(super) fn command_spec() -> ClapCommand {
20    ClapCommand::new("envq")
21        .disable_help_flag(true)
22        .disable_help_subcommand(true)
23        .disable_version_flag(true)
24        .arg(
25            Arg::new("help")
26                .short('h')
27                .long("help")
28                .action(ArgAction::SetTrue),
29        )
30        .arg(
31            Arg::new("version")
32                .long("version")
33                .action(ArgAction::SetTrue),
34        )
35        .arg(Arg::new("quiet").long("quiet").action(ArgAction::SetTrue))
36        .subcommand(command_with_positionals(CommandName::Get, ["KEY", "PATH"]))
37        .subcommand(mutation_command_with_positionals(
38            CommandName::Set,
39            ["KEY", "VALUE", "PATH"],
40        ))
41        .subcommand(mutation_command_with_positionals(
42            CommandName::Clear,
43            ["KEY", "PATH"],
44        ))
45        .subcommand(mutation_command_with_positionals(
46            CommandName::Unset,
47            ["KEY", "PATH"],
48        ))
49        .subcommand(mutation_command_with_positionals(
50            CommandName::Remove,
51            ["KEY", "PATH"],
52        ))
53        .subcommand(command_with_positionals(CommandName::Has, ["KEY", "PATH"]))
54        .subcommand(list_command_with_positionals())
55        .subcommand(
56            subcommand(CommandName::Completion).arg(
57                Arg::new("SHELL")
58                    .required(true)
59                    .value_parser(PossibleValuesParser::new(COMPLETION_SHELLS)),
60            ),
61        )
62        .subcommand(
63            subcommand(CommandName::Help)
64                .arg(Arg::new("COMMAND").value_parser(PossibleValuesParser::new(COMMAND_NAMES))),
65        )
66}

68/// Clap parser for options that appear after mutation operands.

101fn command_with_positionals<const N: usize>(
102    command_name: CommandName,
103    positionals: [&'static str; N],
104) -> ClapCommand {
105    positionals
106        .into_iter()
107        .fold(subcommand(command_name), |command, name| {
108            command.arg(positional_arg(name).required(true))
109        })
110}
111
112fn mutation_command_with_positionals<const N: usize>(
113    command_name: CommandName,
114    positionals: [&'static str; N],
115) -> ClapCommand {
116    add_mutation_options(command_with_positionals(command_name, positionals))
117}
118
119fn list_command_with_positionals() -> ClapCommand {
120    add_list_options(command_with_positionals(CommandName::List, ["PATH"]))
121}
122
123fn subcommand(command_name: CommandName) -> ClapCommand {
124    ClapCommand::new(command_name.as_str())
125        .disable_help_flag(true)
126        .disable_version_flag(true)
127}
128
129fn positional_arg(name: &'static str) -> Arg {
130    let argument = Arg::new(name);
131    if name == "PATH" {
132        argument.value_hint(ValueHint::FilePath)
133    } else {
134        argument
135    }
136}

138fn add_mutation_options(command: ClapCommand) -> ClapCommand {
139    command
140        .arg(
141            Arg::new("stdout")
142                .long("stdout")
143                .action(ArgAction::SetTrue)
144                .group("output"),
145        )
146        .arg(
147            Arg::new("diff")
148                .long("diff")
149                .action(ArgAction::SetTrue)
150                .group("output"),
151        )
152        .arg(Arg::new("check").long("check").action(ArgAction::SetTrue))
153        .group(ArgGroup::new("output").multiple(false))
154}
155
156fn add_list_options(command: ClapCommand) -> ClapCommand {
157    command
158        .arg(
159            Arg::new("json")
160                .long("json")
161                .action(ArgAction::SetTrue)
162                .group("format"),
163        )
164        .arg(
165            Arg::new("yaml")
166                .long("yaml")
167                .action(ArgAction::SetTrue)
168                .group("format"),
169        )
170        .arg(
171            Arg::new("names")
172                .long("names")
173                .action(ArgAction::SetTrue)
174                .group("format"),
175        )
176        .arg(Arg::new("unique").long("unique").action(ArgAction::SetTrue))
177        .group(ArgGroup::new("format").multiple(false))
178}
