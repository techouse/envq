7/// Top-level usage prefix shared by parser errors.

10const COMMAND_USAGE: &str = "commands:
11  envq [--version] [--quiet]
12  get KEY PATH
13  set KEY VALUE PATH [--stdout|--diff] [--check]
14  set KEY - PATH [--stdout|--diff] [--check]
15  clear KEY PATH [--stdout|--diff] [--check]
16  unset KEY PATH [--stdout|--diff] [--check]
17  remove KEY PATH [--stdout|--diff] [--check]
18  has KEY PATH
19  list PATH [--json|--yaml|--names] [--unique]
20  completion {bash,zsh,fish,powershell,pwsh}
21  help [COMMAND]
22";

24const REMOVE_HELP: &str =
25    "usage: envq remove KEY PATH [--stdout|--diff] [--check]\n\nAlias for unset.\n";
26const HELP_HELP: &str = "usage: envq help [COMMAND]\n\nPrint command-specific help.\n";

28/// Returns top-level or command-specific help output.
29pub(super) fn help_result(topic: Option<&str>) -> CliResult {
30    let topic = match topic {
31        Some(topic) => topic,
32        None => {
33            return CliResult {
34                exit_code: ExitCode::Success,
35                stdout: top_help(),
36                stderr: Vec::new(),
37            };
38        }
39    };
40
41    let help_text = match topic {
42        "get" => {
43            "usage: envq get KEY PATH\n\nPrint the first matching value exactly, without adding a newline.\nReturns exit code 2 when KEY is absent.\n"
44        }
45        "set" => {
46            "usage: envq set KEY VALUE PATH [--stdout|--diff] [--check]\n       envq set KEY - PATH [--stdout|--diff] [--check]\n\nSet KEY to VALUE, updating the first match or appending a new binding.\nUse VALUE '-' to read the value from stdin exactly.\nTrailing --stdout prints the rendered file without writing.\nTrailing --diff prints a unified diff without writing.\nTrailing --check exits 4 when the file would change and never writes.\n"
47        }
48        "clear" => {
49            "usage: envq clear KEY PATH [--stdout|--diff] [--check]\n\nKeep or create KEY and set it to an empty value.\nSupports the same trailing output and check options as set.\n"
50        }
51        "unset" => {
52            "usage: envq unset KEY PATH [--stdout|--diff] [--check]\n\nRemove all matching bindings for KEY.\nReturns exit code 2 when KEY is absent.\n"
53        }
54        "remove" => REMOVE_HELP,
55        "has" => {
56            "usage: envq has KEY PATH\n\nPrint nothing. Exit 0 if KEY exists, or 2 if KEY is absent.\n"
57        }
58        "list" => {
59            "usage: envq list PATH [--json|--yaml|--names] [--unique]\n\nPrint bindings in file order as KEY<TAB>VALUE by default.\n--json prints a JSON array of {\"key\", \"value\"} objects.\n--yaml prints a dependency-free YAML sequence.\n--names prints only keys, one per line.\n--unique keeps the first binding for each key and drops later duplicates.\n"
60        }
61        "completion" => {
62            "usage: envq completion {bash,zsh,fish,powershell,pwsh}\n\nPrint a shell completion script to stdout.\n"
63        }
64        "help" => HELP_HELP,
65        _ => {
66            return CliResult {
67                exit_code: ExitCode::GeneralError,
68                stdout: Vec::new(),
69                stderr: format!("envq: unknown help topic: {topic}\n").into_bytes(),
70            };
71        }
72    };
73
74    CliResult {
75        exit_code: ExitCode::Success,
76        stdout: help_text.as_bytes().to_vec(),
77        stderr: Vec::new(),
78    }
79}

81/// Formats a contract-compatible operand-count usage error.
82pub(super) fn command_usage_error(command_name: CommandName) -> Vec<u8> {
83    format!(
84        "{USAGE}envq: error: expected operands: envq {}\n",
85        command_signature(command_name)
86    )
87    .into_bytes()
88}
89
90/// Returns top-level help output.
91pub(super) fn top_help() -> Vec<u8> {
92    format!(
93        "{USAGE}
94Read and edit .env files deterministically.
95
96positional arguments:
97  command     command to run
98  ...         command operands
99
100options:
101  -h, --help  show this help message and exit
102  --version   show program's version number and exit
103  --quiet     suppress command diagnostics on stderr
104
105{COMMAND_USAGE}"
106    )
107    .into_bytes()
108}

110fn command_signature(command_name: CommandName) -> &'static str {
111    match command_name {
112        CommandName::Get => "get KEY PATH",
113        CommandName::Set => "set KEY VALUE PATH",
114        CommandName::Clear => "clear KEY PATH",
115        CommandName::Unset => "unset KEY PATH",
116        CommandName::Remove => "remove KEY PATH",
117        CommandName::Has => "has KEY PATH",
118        CommandName::List => "list PATH",
119        CommandName::Completion => "completion {bash,zsh,fish,powershell,pwsh}",
120        CommandName::Help => "help [COMMAND]",
121    }
122}
