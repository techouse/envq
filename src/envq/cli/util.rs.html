5/// Converts a key operand to a string for later key validation.
6pub(super) fn key_operand(argument: &OsStr) -> String {
7    match std::str::from_utf8(&os_bytes(argument)) {
8        Ok(value) => value.to_owned(),
9        Err(_error) => os_string_lossy(argument),
10    }
11}
12
13/// Formats a missing-key diagnostic.
14pub(super) fn key_not_found_message(key: &str) -> Vec<u8> {
15    format!("envq: {key}: key not found\n").into_bytes()
16}

18/// Formats a path-related OS diagnostic with normalized wording for common errors.
19pub(super) fn os_error_message(path: &Path, error: &io::Error) -> Vec<u8> {
20    format!("envq: {}: {}\n", path.display(), os_error_detail(error)).into_bytes()
21}
22
23/// Formats a stdin read diagnostic.
24pub(super) fn stdin_error_message(error: &io::Error) -> Vec<u8> {
25    format!("envq: stdin: {}\n", os_error_detail(error)).into_bytes()
26}
27
28fn os_error_detail(error: &io::Error) -> String {
29    match error.kind() {
30        io::ErrorKind::NotFound => "No such file or directory".to_owned(),
31        io::ErrorKind::PermissionDenied => "Permission denied".to_owned(),
32        io::ErrorKind::AlreadyExists => "File exists".to_owned(),
33        io::ErrorKind::IsADirectory => "Is a directory".to_owned(),
34        _ => error.to_string(),
35    }
36}

38#[cfg(unix)]
39pub(super) fn os_bytes(value: &OsStr) -> Vec<u8> {
40    use std::os::unix::ffi::OsStrExt;
41
42    value.as_bytes().to_vec()
43}
44
45#[cfg(not(unix))]
46pub(super) fn os_bytes(value: &OsStr) -> Vec<u8> {
47    value.to_string_lossy().as_bytes().to_vec()
48}
49
50pub(super) fn os_string_lossy(value: &OsStr) -> String {
51    String::from_utf8_lossy(&os_bytes(value)).into_owned()
52}
